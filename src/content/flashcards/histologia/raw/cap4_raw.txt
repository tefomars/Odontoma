        continue

    lines = raw_path.read_text(encoding="utf-8-sig").splitlines()

    rows = []

    for line in lines:
        line = line.strip()

        if not line:
            continue

        if line.lower() == "chapter\tsubtopic\tfront\tback":
            continue

        parts = line.split("\t")

        if len(parts) < 4:
            continue

        chapter = parts[0].strip()
        subtopic = parts[1].strip()
        front = parts[2].strip()
        back = "\t".join(parts[3:]).strip()

        if not chapter or not subtopic or not front or not back:
            continue

        rows.append([chapter, subtopic, front, back])

    with csv_path.open("w", encoding="utf-8-sig", newline="") as f:
        writer = csv.writer(f)
        writer.writerow(expected_header)
        writer.writerows(rows)

    print(f"{csv_name}: {len(rows)} flashcards")

print("CSV listos.")

from pathlib import Path
import csv
import json

raw_dir = Path("src/content/flashcards/histologia/raw")
out_path = Path("src/content/flashcards/histologia/cards.ts")

files = [
    raw_dir / "cap4.csv",
    raw_dir / "cap5.csv",
]

def read_csv(path):
    rows = []

    with path.open("r", encoding="utf-8-sig", newline="") as f:
        reader = csv.DictReader(f)

        for row in reader:
            chapter = (row.get("chapter") or "").strip()
            subtopic = (row.get("subtopic") or "").strip()
            front = (row.get("front") or "").strip()
            back = (row.get("back") or "").strip()

            if not chapter or not subtopic or not front or not back:
                continue

            rows.append({
                "chapter": chapter,
                "subtopic": subtopic,
                "front": front,
                "back": back,
            })

    return rows

all_rows = []

for path in files:
    if path.exists():
        all_rows.extend(read_csv(path))

if not all_rows:
    raise SystemExit("No encontré flashcards válidas en cap4.csv o cap5.csv.")

def chapter_number(chapter):
    digits = "".join(ch for ch in chapter if ch.isdigit())
    return digits or "x"

def slug(text):
    text = text.lower()
    replacements = {
        "á": "a",
        "é": "e",
        "í": "i",
        "ó": "o",
        "ú": "u",
        "ü": "u",
        "ñ": "n",
    }

    for old, new in replacements.items():
        text = text.replace(old, new)

    return "".join(
        ch if ch.isalnum() else "-"
        for ch in text
    ).strip("-")

cards = []
seen = set()

for index, row in enumerate(all_rows, start=1):
    key = (
        row["chapter"],
        row["subtopic"],
        row["front"],
        row["back"],
    )

    if key in seen:
        continue

    seen.add(key)

    cap = chapter_number(row["chapter"])
    topic_slug = slug(row["subtopic"])

    cards.append({
        "id": f"histo-cap{cap}-{topic_slug}-{index:04d}",
        "subject": "Histología",
        "book": "Ross",
        "chapter": row["chapter"],
        "topic": row["subtopic"],
        "subtopic": row["subtopic"],
        "front": row["front"],
        "back": row["back"],
        "tags": [
            row["chapter"],
            row["subtopic"],
        ],
    })

content = """export type Flashcard = {
  id: string
  subject: string
  book: string
  chapter: string
  topic: string
  subtopic: string
  front: string
  back: string
  tags: string[]
}

export const histologiaFlashcards: Flashcard[] = """

content += json.dumps(cards, ensure_ascii=False, indent=2)
content += "\n"

out_path.write_text(content, encoding="utf-8")

print(f"Listo: {len(cards)} flashcards integradas en {out_path}")
